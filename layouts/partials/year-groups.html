{{- /* year-groups.html — renders a page collection as year-grouped
blocks, shared by the index, the post listing and the tag pages. Years appear in first-seen order (the collection is date-descending, so the newest year first). Arguments: dict "pages" "start" . */ -}} {{- $posts := .pages -}} {{- $start := .start -}} {{- $years := slice -}} {{- range $posts -}} {{- $y := printf "%d" .Date.Year -}} {{- if not (in $years $y) -}} {{- $years = append $y $years -}} {{- end -}} {{- end -}} {{- range $i, $year := $years -}} {{- $yp := where $posts "Date.Year" (int $year) -}}

{{ $year }} {{ len $yp }} post{{ if ne (len $yp) 1 }}s{{ end }}

    {{- range $yp }} {{ .Render "li" }} {{- end }}
{{- end -}}
